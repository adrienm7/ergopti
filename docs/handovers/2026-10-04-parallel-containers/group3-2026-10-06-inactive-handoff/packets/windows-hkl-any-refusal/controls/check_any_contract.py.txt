"""Private source preservation controls; actual AHK refusal cases remain unexecuted."""

from pathlib import Path
import difflib
import json

BASE = Path(__file__).resolve().parent
REL = "static/ergopti_plus/windows/adapters/key_state.ahk"
TEST = "static/ergopti_plus/windows/tests/unit/test_layout_digit_row_probe.ahk"


def check(source):
    addition = source.split(
        "KS_NativeKeyLevel(Vk, Scancode, Hkl, Shift, Caps, ReadFn := 0) {", 1
    )[1]
    assert addition.count("} catch Any {") == 2
    assert "} catch {" not in addition
    assert addition.count("Observed.Count") == addition.count("Observed.Text") == 1


def main():
    pre = (BASE / "preimages" / REL).read_text(encoding="utf-8-sig")
    post = (BASE / "candidate" / REL).read_text(encoding="utf-8-sig")
    check(post)
    assert post.replace("} catch Any {", "} catch {") == pre
    try:
        check(pre)
    except AssertionError:
        print(
            "EXPECTED RED: predecessor only catches Error at both new external boundaries"
        )
    else:
        raise AssertionError("Missing Any boundary was accepted")
    for ordinal in [1, 2]:
        prefix, body = post.split(
            "KS_NativeKeyLevel(Vk, Scancode, Hkl, Shift, Caps, ReadFn := 0) {", 1
        )
        index = [i for i in range(len(body)) if body.startswith("} catch Any {", i)][
            ordinal - 1
        ]
        mutated = (
            prefix
            + "KS_NativeKeyLevel(Vk, Scancode, Hkl, Shift, Caps, ReadFn := 0) {"
            + body[:index]
            + body[index:].replace("} catch Any {", "} catch {", 1)
        )
        try:
            check(mutated)
        except AssertionError:
            print("EXPECTED RED: omitted Any boundary " + str(ordinal))
        else:
            raise AssertionError("Partial Any omission survived")
    before = (BASE / "preimages" / TEST).read_text(encoding="utf-8-sig")
    after = (BASE / "candidate" / TEST).read_text(encoding="utf-8-sig")
    assert (
        before.count('Test("native HKL number row:')
        == after.count('Test("native HKL number row:')
        == 6
    )
    removed = [
        line
        for line in difflib.ndiff(before.splitlines(), after.splitlines())
        if line.startswith("- ")
    ]
    assert removed == [], "No old assertion or test-source line may be removed"
    assert 'return "escaped native observation boundary"' in after
    assert "throw Value" in after
    assert 'for Value in ["private non-Error descriptor refusal", 71]' in after
    assert "_NHKL_NoEscapingValue(() => KS_NativeNumberRowLevels" in after
    for rel in [REL, TEST]:
        data = (BASE / "candidate" / rel).read_bytes()
        assert data.startswith(b"\xef\xbb\xbf") and b"\r" not in data
    result = {
        "source": "passed",
        "predecessor": "expected failure",
        "omitted_Any_boundaries": 2,
        "old_test_lines": "all retained in order",
        "registered_cases": 6,
        "native_behavior": "unexecuted",
    }
    (BASE / "source-controls.json").write_text(json.dumps(result, indent=2) + "\n")
    print(
        "PASS: two Any boundaries; no old test-line removal; registered native cases remain six"
    )
    print("UNEXECUTED: actual non-Error native ReadFn/MapFn/own-getter throw behavior")


if __name__ == "__main__":
    main()

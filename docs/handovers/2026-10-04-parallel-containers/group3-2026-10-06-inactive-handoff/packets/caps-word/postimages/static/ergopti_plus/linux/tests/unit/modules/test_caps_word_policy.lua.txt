--- tests/unit/modules/test_caps_word_policy.lua

--- ==============================================================================
--- MODULE: Shared Caps Word Casing Ownership
--- DESCRIPTION:
--- Independent legacy title-word vectors, sealed native probes and actual Linux
--- shortcut-manager composition. No physical native input is claimed.
--- ==============================================================================

local NativeLoop=require("luv")
local helpers=require("tests.helpers")
local Casing=require("tap_hold.caps_word")
helpers.describe("one versioned Caps Word state owner",function()
	helpers.it("preserves title-case first-letter and repeated-word semantics",function()
		local owner=Casing.new();helpers.assert_eq(owner.toggle_title(),true)
		helpers.assert_eq(owner.process_title("é"),"É");helpers.assert_eq(owner.process_title("b"),nil)
		helpers.assert_eq(owner.process_title(" "),nil);helpers.assert_eq(owner.process_title("a"),"A")
		helpers.assert_eq(owner.process_title("."),nil);helpers.assert_eq(owner.process_title("n"),"N")
		helpers.assert_eq(owner.active(),true)
	end)
	helpers.it("does not treat invalid or empty text as an invented word boundary",function()
		local owner=Casing.new();owner.toggle_title()
		helpers.assert_eq(owner.process_title(""),nil);helpers.assert_eq(owner.process_title("\255"),nil)
		helpers.assert_eq(owner.triggered(),false);helpers.assert_eq(owner.process_title("a"),"A")
	end)
	helpers.it("keeps the original already-uppercase and digit title-word consumption",function()
		local owner=Casing.new();owner.toggle_title()
		helpers.assert_eq(owner.process_title("A"),nil);helpers.assert_eq(owner.process_title("b"),nil)
		owner.process_title(" ");helpers.assert_eq(owner.process_title("1"),nil)
		helpers.assert_eq(owner.process_title("b"),nil)
	end)
	helpers.it("prevents the legacy on-char observer from duplicating native uppercase output",function()
		local owner=Casing.new();local accepted,claim=owner.toggle_native();helpers.assert_eq(accepted,true)
		helpers.assert_eq(owner.process_title("a"),nil)
		local receipt,text=owner.probe_native(claim,"a");helpers.assert_eq(text,"A")
		helpers.assert_eq(owner.commit_native(claim,receipt),true);helpers.assert_eq(owner.current_native(claim),true)
	end)
	helpers.it("does not commit a probe after pointer cancellation",function()
		local owner=Casing.new();local _,claim=owner.toggle_native()
		local receipt,text=owner.probe_native(claim,"a");helpers.assert_eq(text,"A")
		owner.pointer_activity(claim);helpers.assert_eq(owner.commit_native(claim,receipt),false)
		helpers.assert_eq(owner.active(),false)
	end)
	helpers.it("never lets an old native claim clear newer title-word intent",function()
		local owner=Casing.new();local _,claim=owner.toggle_native()
		owner.toggle_title();owner.toggle_title()
		helpers.assert_eq(owner.mode(),"title_words");helpers.assert_eq(owner.active(),true)
		helpers.assert_eq(owner.retire_native(claim),true);helpers.assert_eq(owner.active(),true)
		helpers.assert_eq(owner.process_title("a"),"A")
	end)
	helpers.it("rejects foreign or cloned probes without publishing guessed state",function()
		local owner=Casing.new();local _,claim=owner.toggle_native()
		local receipt=owner.probe_native(claim," ")
		helpers.assert_eq(owner.commit_native({},receipt),false)
		helpers.assert_eq(owner.commit_native(claim,{}),false)
		helpers.assert_eq(owner.current_native(claim),true)
		helpers.assert_eq(owner.commit_native(claim,receipt),true)
		helpers.assert_eq(owner.commit_native(claim,receipt),false);helpers.assert_eq(owner.active(),false)
	end)
	helpers.it("does not forge a title snapshot while native upper-word ownership is active",function()
		local owner=Casing.new();local _,claim=owner.toggle_native()
		helpers.assert_eq(owner.title_snapshot(),nil)
		owner.retire_native(claim);local snapshot=owner.title_snapshot()
		helpers.assert_eq(snapshot.active,false);helpers.assert_eq(snapshot.triggered,false)
		helpers.assert_eq(owner.restore_title(true,true),true)
		helpers.assert_eq(owner.process_title("a"),nil)
		helpers.assert_eq(owner.restore_title(true,false),true);helpers.assert_eq(owner.process_title("a"),"A")
	end)
	helpers.it("rejects stale probes even if a different mode returns to native uppercase",function()
		local owner=Casing.new();local _,claim=owner.toggle_native();local receipt=owner.probe_native(claim,"a")
		owner.toggle_title();local _,successor=owner.toggle_native()
		helpers.assert_eq(owner.current_native(successor),true)
		helpers.assert_eq(owner.commit_native(claim,receipt),false)
	end)
end)
helpers.describe("actual Linux title-word manager and native casing port",function()
	helpers.it("shares one owner while preserving menu toggles and detached title snapshots",function()
		local manager=require("modules.shortcuts.manager")
		manager.init({persist=true,config_path="/controlled/config.toml",enabled=true})
		local native=manager.native_casing_owner();helpers.assert_eq(native,manager.native_casing_owner())
		helpers.assert_eq(manager.toggle_caps_word(),true);helpers.assert_eq(manager.process_caps_word("a"),"A")
		helpers.assert_eq(manager.process_caps_word("b"),nil)
		local token={};helpers.assert_eq(manager.acquire_configuration(token),true)
		local snapshot=manager.configuration_snapshot(token);helpers.assert_eq(snapshot.caps_word_active,true)
		helpers.assert_eq(snapshot.caps_word_triggered,true);helpers.assert_eq(native.toggle_native(),false)
		helpers.assert_eq(manager.release_configuration(token),true)
		manager.toggle_caps_word();local accepted,claim=native.toggle_native();helpers.assert_eq(accepted,true)
		helpers.assert_eq(native.current_native(claim),true);helpers.assert_eq(manager.process_caps_word("a"),nil)
		helpers.assert_eq(manager.acquire_configuration(token),true)
		helpers.assert_eq(manager.configuration_snapshot(token),nil,"native output ownership is not a guessed legacy boolean snapshot")
		helpers.assert_eq(native.current_native(claim),false)
		helpers.assert_eq(manager.release_configuration(token),true)
		native.retire_native(claim);manager.toggle_caps_word()
		helpers.assert_eq(native.retire_native(claim),true);helpers.assert_eq(manager.process_caps_word("a"),"A")
	end)
end)

helpers.describe("native casing publication ownership",function()
	helpers.it("cannot apply guessed title booleans over an unretired native upper-word claim",function()
		local manager=require("modules.shortcuts.manager");manager.init({persist=true,config_path="/controlled/config.toml",enabled=true})
		local native=manager.native_casing_owner();local accepted,claim=native.toggle_native();helpers.assert_eq(accepted,true)
		local token={};helpers.assert_eq(manager.acquire_configuration(token),true)
		helpers.assert_eq(manager.apply_configuration(token,{enabled=false,wrap=false,caps_word_active=false,caps_word_triggered=false}),false)
		helpers.assert_eq(manager.release_configuration(token),true);helpers.assert_eq(native.current_native(claim),true)
		native.retire_native(claim);helpers.assert_eq(manager.acquire_configuration(token),true)
		helpers.assert_eq(manager.apply_configuration(token,{enabled=true,wrap=true,caps_word_active=true,caps_word_triggered=false}),true)
		helpers.assert_eq(manager.release_configuration(token),true);helpers.assert_eq(manager.process_caps_word("n"),"N")
	end)
end)

helpers.describe("native casing probe revision",function()
	helpers.it("cannot commit an outer plan after a nested probe commits the same native claim",function()
		local owner=Casing.new();local _,claim=owner.toggle_native()
		local outer=owner.probe_native(claim,"a");local nested=owner.probe_native(claim,"b")
		helpers.assert_eq(owner.commit_native(claim,nested),true)
		helpers.assert_eq(owner.commit_native(claim,outer),false)
		helpers.assert_eq(owner.current_native(claim),true)
	end)
end)

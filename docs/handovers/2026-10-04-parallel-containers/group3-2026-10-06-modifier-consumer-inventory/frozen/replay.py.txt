"""Prepare and replay the immutable consumer diagnostics in a new private snapshot."""
from pathlib import Path
import argparse
import hashlib
import json
import shutil
import subprocess

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("repository", type=Path)
parser.add_argument("private_directory", type=Path)
parser.add_argument("interpreters", nargs="+", type=Path)
args = parser.parse_args()
packet = Path(__file__).resolve().parent
repo = args.repository.resolve()
work = args.private_directory.resolve()
if work == repo or repo in work.parents:
    raise SystemExit("The replay directory must be outside the repository checkout")
manifest = json.loads((packet / "source-pins.json").read_text())
work.mkdir(parents=True, exist_ok=False)
snapshot = work / "snapshot"
sha = lambda value: hashlib.sha256(value).hexdigest()
for row in manifest["snapshot_sources"]:
    data = (repo / row["path"]).read_bytes()
    if sha(data) != row["sha256"]:
        raise SystemExit("Source preimage changed: " + row["path"])
    target = snapshot / row["path"]
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
for row in manifest["overlays"]:
    data = (repo / row["inactive_postimage"]).read_bytes()
    if sha(data) != row["reviewed_postimage"]:
        raise SystemExit("Inactive overlay changed: " + row["inactive_postimage"])
    (snapshot / row["path"]).write_bytes(data)
for row in manifest["predecessor_receipts"]:
    if sha((repo / row["path"]).read_bytes()) != row["sha256"]:
        raise SystemExit("Inactive predecessor receipt changed: " + row["path"])
watchdog = snapshot / manifest["watchdog_path"]
overlap = (repo / manifest["overlap_path"]).read_bytes()
if sha(overlap) != manifest["overlap_sha256"]:
    raise SystemExit("Original overlap assertion changed")
original = watchdog.read_bytes()
watchdog.write_bytes(original + overlap + (packet / "controls.lua").read_bytes())
if not watchdog.read_bytes().startswith(original + overlap):
    raise SystemExit("Original assertions were not preserved")
(work / "counterfactuals").mkdir()
for row in manifest["counterfactuals"]:
    source = (snapshot / row["source_path"]).read_text()
    for mutation in row["mutations"]:
        if source.count(mutation["before"]) != 1:
            raise SystemExit("Counterfactual anchor changed: " + row["name"])
        source = source.replace(mutation["before"], mutation["after"], 1)
    (work / "counterfactuals" / (row["name"] + ".lua")).write_text(source)
shutil.copyfile(packet / "run.lua", work / "run.lua")
results = []
for interpreter in args.interpreters:
    for selected in ["current", "skip-only-up", "weaken-baseline", "legacy-retirement", "forbid-retirement"]:
        output = work / (interpreter.name + "-" + selected + ".log")
        with output.open("wb") as stream:
            run = subprocess.run([str(interpreter), str(work / "run.lua"), str(work), selected],
                                 cwd=work, stdout=stream, stderr=subprocess.STDOUT, check=False)
        lines = output.read_text().splitlines()
        expected = manifest["expected"][selected]
        final = f"AGGREGATE_CONSUMER_INVENTORY {expected['passed']} {expected['failed']}"
        if lines[-1] != final or run.returncode != expected["exit_code"]:
            raise SystemExit("Unexpected diagnostic result: " + output.name)
        results.append({"interpreter": interpreter.name, "selection": selected,
                        "exit_code": run.returncode, "final": lines[-1], "log": str(output)})
(work / "results.json").write_text(json.dumps(results, indent=2) + "\n")
print(json.dumps({"status": "EXPECTED_DIAGNOSTIC_OUTCOMES_REPRODUCED", "results": results}, indent=2))

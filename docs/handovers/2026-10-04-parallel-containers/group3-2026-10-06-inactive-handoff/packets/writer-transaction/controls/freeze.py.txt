from pathlib import Path
import hashlib,json,difflib,re
root=Path('/workspace/.ergopti-cloud/preparations/group3-linux-output-transaction')
base=Path('/workspace/.ergopti-cloud/preparations/group3-linux-physical-output-native-owner')
paths=['static/ergopti_plus/linux/adapters/uinput_writer.lua','static/ergopti_plus/linux/tests/unit/adapters/test_uinput_writer.lua']
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
patch=[];sources=[]
for rel in paths:
    pre=root/'preimages'/rel;post=root/'candidate'/rel
    assert pre.read_bytes()==(base/'candidate'/rel).read_bytes()
    if '/tests/' in rel:assert post.read_bytes().startswith(pre.read_bytes())
    patch.extend(difflib.unified_diff(pre.read_text().splitlines(True),post.read_text().splitlines(True),fromfile='a/'+rel,tofile='b/'+rel))
    sources.append({'path':rel,'preimage':sha(pre),'postimage':sha(post)})
(root/'candidate.patch').write_text(''.join(patch))
results=[]
for abi in ['luajit','lua54']:
    for which in ['before','after']:
        p=root/(abi+'-'+which+'.log');line=p.read_text().strip().splitlines()[-1]
        m=re.fullmatch(r'OUTPUT_OWNER (\d+) (\d+)',line);assert m,line
        results.append({'abi':abi,'source':which,'passed':int(m[1]),'failed':int(m[2]),'log':p.name,'sha256':sha(p)})
(root/'results.json').write_text(json.dumps(results,indent=2)+'\n')
(root/'README.md').write_text('''# Private acknowledged Linux output transaction successor

artifacts=[]
for p in sorted(root.rglob('*')):
    if p.is_file() and p.name!='receipt.json':
        artifacts.append({'path':str(p.relative_to(root)),'sha256':sha(p),'bytes':p.stat().st_size})
receipt={'status':'PRIVATE_NOT_ADMITTED','foundation_receipt':sha(base/'receipt.json'),'owned_paths':sources,'artifacts':artifacts,'results':results,'native_execution':'NOT_RUN','physical_delivery_available':False}

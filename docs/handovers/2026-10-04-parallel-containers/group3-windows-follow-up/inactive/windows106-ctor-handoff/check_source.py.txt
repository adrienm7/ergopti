def bodies(text):
    result={}
    for m in re.finditer(r'^([A-Za-z_]\w*)\(',text,re.M):
        # A function header can span lines; its first non-string block brace
        # follows the matching parameter list.
        pos=m.end()-1; par=0; quote=None; escaped=False; brace=None
        while pos<len(text):
            c=text[pos]
            if quote:
                if escaped: escaped=False
                elif c=='`': escaped=True
                elif c==quote: quote=None
            elif c in "\"'": quote=c
            elif c=='(': par+=1
            elif c==')':
                par-=1
                if par==0:
                    p=pos+1
                    while p<len(text) and text[p].isspace(): p+=1
                    if p<len(text) and text[p]=='{': brace=p
                    break
            pos+=1
        if brace is None: continue
        pos=brace; depth=0; quote=None; escaped=False
        while pos<len(text):
            c=text[pos]
            if quote:
                if escaped: escaped=False
                elif c=='`': escaped=True
                elif c==quote: quote=None
            elif c in "\"'": quote=c
            elif c==';':
                p=text.find('\n',pos)
                if p<0: break
                pos=p
            elif c=='{': depth+=1
            elif c=='}':
                depth-=1
                if depth==0:
                    result[m.group(1)]=text[m.start():pos+1]
                    break
            pos+=1
    return result
